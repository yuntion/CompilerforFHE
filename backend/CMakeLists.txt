cmake_minimum_required(VERSION 3.24)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(CrossFHEModel LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CROSSFHE_BACKEND "openfhe-cross" CACHE STRING "Backend")
  set(CROSSFHE_MODEL_HEADER "" CACHE FILEPATH "Generated model header")
  if(NOT EXISTS "${CROSSFHE_MODEL_HEADER}")
    message(FATAL_ERROR "CROSSFHE_MODEL_HEADER must name a compiled model")
  endif()
  if(CROSSFHE_BACKEND STREQUAL "seal-ckks")
    add_subdirectory(seal)
  elseif(CROSSFHE_BACKEND STREQUAL "openfhe-cross")
    add_subdirectory(openfhe)
  else()
    message(FATAL_ERROR "Unsupported backend; GPU code generation is not implemented")
  endif()
  target_compile_definitions(model-runner PRIVATE CROSSFHE_MODEL_HEADER="${CROSSFHE_MODEL_HEADER}")
  set_target_properties(model-runner PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
else()
  if(NOT CROSSFHE_SEAL_SOURCE)
    message(FATAL_ERROR "Set CROSSFHE_SEAL_SOURCE; scripts/bootstrap.py fetches the pinned source")
  endif()
  set(SEAL_BUILD_DEPS OFF CACHE BOOL "" FORCE)
  set(SEAL_USE_MSGSL OFF CACHE BOOL "" FORCE)
  set(SEAL_USE_ZLIB OFF CACHE BOOL "" FORCE)
  set(SEAL_USE_ZSTD OFF CACHE BOOL "" FORCE)
  set(SEAL_USE_INTEL_HEXL OFF CACHE BOOL "" FORCE)
  set(SEAL_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(SEAL_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  if(MINGW)
    set(SEAL_USE_ALIGNED_ALLOC OFF CACHE BOOL "" FORCE)
  endif()
  add_subdirectory("${CROSSFHE_SEAL_SOURCE}" "${CMAKE_BINARY_DIR}/seal")
  if(MINGW)
    target_link_libraries(seal PUBLIC bcrypt)
  endif()
endif()
